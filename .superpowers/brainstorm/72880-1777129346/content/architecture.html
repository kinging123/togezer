lib/supabase.ts — Clerk JWT bridge
// Called before every Supabase request
import { createClient } from '@supabase/supabase-js'
import { useSession } from '@clerk/expo'

export function useSupabase() {
  const { session } = useSession()

  return createClient(URL, ANON_KEY, {
    auth: { persistSession: false },
    global: {
      headers: {
        Authorization: async () => {
          const token = await session
            ?.getToken({ template: 'supabase' })
          return `Bearer ${token}`
        }
      }
    }
  })
}

import { QueryClient } from '@tanstack/react-query'

export const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 1000 * 60 * 2,  // 2 min
      gcTime:    1000 * 60 * 10, // 10 min
      retry: 1,
    },
  },
})
features/habits/hooks — query key factory
// Centralised keys — no magic strings anywhere
export const habitKeys = {
  all:    () => ['habits']                  as const,
  list:   () => ['habits', 'list']          as const,
  detail: (id: string) =>
          ['habits', 'detail', id]          as const,
}

export function useHabits() {
  const sb = useSupabase()
  return useQuery({
    queryKey: habitKeys.list(),
    queryFn: () => sb
      .from('habits')
      .select('*')
      .eq('is_archived', false)
      .order('created_at')
  })
}

RLS policy pattern (same for every table)
-- profiles: can only read/write your own row
CREATE POLICY "own profile" ON profiles
  USING ( id = auth.jwt() ->> 'sub' );

-- habits: only owner sees their habits
CREATE POLICY "own habits" ON habits
  USING ( user_id = auth.jwt() ->> 'sub' );

-- check_ins: owner OR mutual friend can read
CREATE POLICY "friends can read check_ins" ON check_ins
  FOR SELECT USING (
    user_id = auth.jwt() ->> 'sub'
    OR EXISTS (
      SELECT 1 FROM friendships
      WHERE (user_a_id = auth.jwt() ->> 'sub' AND user_b_id = check_ins.user_id)
         OR (user_b_id = auth.jwt() ->> 'sub' AND user_a_id = check_ins.user_id)
    )
  );
One Supabase client instance per session, not per component. The useSupabase() hook shown above is simplified — in practice you'll memoize the client (e.g. in a context provider) so it isn't recreated on every render. The pattern is the same; the client just lives higher in the tree.
